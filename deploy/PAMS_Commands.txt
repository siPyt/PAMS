------------------------------  RAW BACnet TOOLS  ------------------------------
  (used under the hood by pams_control.py; MS/TP env must be set)

  export BACNET_DATALINK=mstp BACNET_IFACE=/dev/ttyUSB0 \
         BACNET_MSTP_BAUD=38400 BACNET_MSTP_MAC=45 BACNET_MAX_MASTER=127
  ~/bacnet-stack/bin/bacwi                              # Who-Is (discover)
  ~/bacnet-stack/bin/bacepics 1234                      # dump device 1234
  ~/bacnet-stack/bin/bacrp 1234 analog-input 1 present-value   # read
  ~/bacnet-stack/bin/bacwp 1234 analog-value 50 present-value 8 -1 4 87.5  # write
